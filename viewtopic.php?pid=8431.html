    "enforce_dims_coord" CHECK (ndims(coord) = 2)
    "enforce_geotype_coord" CHECK (geometrytype(coord) = 'POINT'::text OR coord IS NULL)
    "enforce_srid_coord" CHECK (srid(coord) = 4326)
